Restaurant Graph Database Ð import.txt codeThis document contains the Cypher code for importing the restaurant relational database from CSV files into Neo4j, including constraints, an index, nodes, relationships, and verification queries.//Task 2: Graph Database Implementation// 1. CREATE UNIQUE CONSTRAINTSCREATE CONSTRAINT customer_id_unique IF NOT EXISTSFOR (c:Customer)REQUIRE c.CustomerID IS UNIQUE;CREATE CONSTRAINT staff_id_unique  IF NOT EXISTSFOR (s:Staff)REQUIRE s.StaffID IS UNIQUE;CREATE CONSTRAINT order_id_unique IF NOT EXISTSFOR (o:Order)REQUIRE o.OrderID IS UNIQUE;CREATE CONSTRAINT category_id_unique IF NOT EXISTSFOR (c:MenuCategory)REQUIRE c.CategoryID IS UNIQUE;CREATE CONSTRAINT menuitem_id_unique IF NOT EXISTSFOR (m:MenuItem)REQUIRE m.MenuItemID IS UNIQUE;CREATE CONSTRAINT payment_id_unique IF NOT EXISTSFOR (p:Payment)REQUIRE p.PaymentID IS UNIQUE;// 2. CREATE INDEXCREATE INDEX menuitem_name_index IF NOT EXISTSFOR (m:MenuItem)ON (m.ItemName);// 3. IMPORT CUSTOMER NODESLOAD CSV WITH HEADERS FROM Ôfile:///customers.csvÕ AS rowCREATE (:Customer {	CustomerID: toInteger (row.CustomerID),	FirstName: row.FirstName,	LastName: row.LastName,	Phone: row.Phone,	Email: row.Email});// 4. IMPORT STAFF NODESLOAD CSV WITH HEADERS FROM Ôfile:///staff.csvÕ AS rowCREATE (:Staff {	StaffID: toInteger (row.StaffID),	FirstName: row.FirstName,	LastName: row.LastName,	Phone: row.Phone,});// 5. APPLY MANAGER LABELLOAD CSV WITH HEADERS FROM Ôfile:///managers.csvÕ AS row MATCH (s:Staff {StaffID: toInteger (row.StaffID)})SET s:Manager;// 6. APPLY SERVICE STAFF  LABELLOAD CSV WITH HEADERS FROM Ôfile:///service_staff.csvÕ AS row MATCH (s:Staff {StaffID: toInteger (row.StaffID)})SET s:ServiceStaff;// 7. IMPORT MENU CATEGORY NODESLOAD CSV WITH HEADERS FROM Ôfile:///menucategories.csvÕ AS rowCREATE (:MenuCategory {	CategoryID: toInteger(row.CategoryID),	CategoryName: row.CategoryName});// 8. IMPORT MENU ITEM NODESLOAD CSV WITH HEADERS FROM Ôfile:///menu_items.csvÕ AS rowCREATE (:MenuItem {	MenuItemID: toInteger(row.MenuItemID),       CategoryID: toInteger(row.CategoryID),	ItemName: row.ItemName,	Description: row.Description,	Price: toFloat(row.price),	Availability: row.Availability});// 9. IMPORT ORDER NODESLOAD CSV WITH HEADERS FROM Ôfile:///orders.csvÕ AS rowCREATE (:Order {	OrderID: toInteger(row.OrderID),       CustomerID: toInteger(row.CustomerID),       StaffID: toInteger(row.StaffID),       OrderDate: row.OrderDate ,       OrderStatus: row.OrderStatus	});// 10. IMPORT PAYMENT NODESLOAD CSV WITH HEADERS FROM Ôfile:///payments.csvÕ AS rowCREATE (:Payment {	PaymentID: toInteger(row.PaymentID),       OrderID: toInteger(row.OrderID),       Amount: toFloat(row.Amount),	PaymentDate: row.PaymentDate,	PaymentMethod: row.PaymentMethod,	PaymentStatus: row.PaymentStatus});// 11. CREATE CUSTOMER -> ORDER RELATIONSHIPSLOAD CSV WITH HEADERS FROM Ôfile:///orders.csvÕ As rowMATCH (c:Customer {CustomerID: toInteger(row.CustomerID)})MATCH (o:order {orderId:toInteger(row.OrderID)})MERGE (c) Ð [:PLACES] -> (o);// 12. CREATE STAFF -> ORDER RELATIONSHIPSLOAD CSV WITH HEADERS FROM Ôfile:///orders.csvÕ As rowMATCH (s:Staff {StaffID: toInteger(row.StaffID)})MATCH (o:order {orderId:toInteger(row.OrderID)})MERGE (c) Ð [:HANDLES] -> (o);// 13. CREATE MENU ITEM -> CATEGORY RELATIONSHIPSLOAD CSV WITH HEADERS FROM Ôfile:///menu_items.csvÕ As rowMATCH (m:MenuItem {MenuItemID: toInteger(row.MenuItemID)})MATCH (c:MenuCategory {CategoryID:toInteger(row.CategoryID)})MERGE (m) Ð [:BELONGS_TO] -> (c);// 14. CREATE ORDER -> MENU ITEM RELATIONSHIPS//ORDER_ITEM is represented as the CONTAINS relationship.LOAD CSV WITH HEADERS FROM Ôfile:///order_items.csvÕ As rowMATCH (o:Order {OrderID: toInteger(row.orderID)})MATCH (m:MenuITEM {MenuITEMID:toInteger(row.MenuItemID)})MERGE (o) Ð [r:CONTAINS] -> (m);SET r.Quantity = toInteger (row.Quantity),	r.UnitPrice = toFloat (row.UnitPrice);// 15. CREATE ORDER -> PAYMENT RELATIONSHIPSLOAD CSV WITH HEADERS FROM Ôfile:///payments.csvÕ As rowMATCH (o:Order {OrderID: toInteger(row.orderID)})MATCH (p:Payment {PaymentID:toInteger(row.PaymentID)})MERGE (o) Ð [:HAS_PAYMENT] -> (p);// 16. VERIFY IMPORTED NODESMATCH (n)RETURN labels(n) AS NodeLabels,       count (*) As NodeCountORDER BY NodeLabels;// 17. VERIFY IMPORTED RELATIONSHIPSMATCH () Ð[r] -> ()RETURN type(r) AS RelationshipType,       count (*) As RelationshipCountORDER BY RelationshipType;// 18. VERIFY CONSTRAINTSSHOW CONSTRAINTS//19. VERIFY INDEXESSHOW INDEXES;
